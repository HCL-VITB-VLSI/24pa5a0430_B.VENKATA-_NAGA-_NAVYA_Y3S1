// =============================================================
//  Testbench: rr_arbiter_tb
//  Experiment 2 — Fairness Engine
//
//  Features:
//    • 4 directed test cases (single, alternating, all-active,
//      gap/idle)
//    • 200-cycle random stimulus
//    • Golden reference model (mirror pointer + expected grant)
//    • Mismatch counter with $display diagnostics
//    • VCD dump for waveform inspection
// =============================================================

`timescale 1ns/1ps

module rr_arbiter_tb;

    // -------------------------------------------------------
    // Parameters — change WIDTH here to test 4 / 8 / 16
    // -------------------------------------------------------
    parameter integer WIDTH = 4;
    localparam PTR_W = $clog2(WIDTH);

    // -------------------------------------------------------
    // DUT connections
    // -------------------------------------------------------
    reg             clk;
    reg             rst_n;
    reg  [WIDTH-1:0] req;
    wire [WIDTH-1:0] grant;
    wire             grant_valid;

    rr_arbiter #(.WIDTH(WIDTH)) dut (
        .clk        (clk),
        .rst_n      (rst_n),
        .req        (req),
        .grant      (grant),
        .grant_valid(grant_valid)
    );

    // -------------------------------------------------------
    // Clock: 10 ns period
    // -------------------------------------------------------
    initial clk = 0;
    always #5 clk = ~clk;

    // -------------------------------------------------------
    // VCD dump
    // -------------------------------------------------------
    initial begin
        $dumpfile("rr_arbiter_tb.vcd");
        $dumpvars(0, rr_arbiter_tb);
    end

    // -------------------------------------------------------
    // Golden reference model
    // -------------------------------------------------------
    integer          gold_ptr;
    reg [WIDTH-1:0]  gold_grant;

    // Compute golden grant combinationally from gold_ptr + req
    task compute_golden;
        input [WIDTH-1:0] r;
        input integer     ptr_in;
        output [WIDTH-1:0] g;
        integer i, idx;
        begin
            g = {WIDTH{1'b0}};
            for (i = 0; i < WIDTH; i = i + 1) begin
                idx = (ptr_in + i) % WIDTH;
                if (r[idx] && g == {WIDTH{1'b0}}) begin
                    g[idx] = 1'b1;
                end
            end
        end
    endtask

    // -------------------------------------------------------
    // Error counter
    // -------------------------------------------------------
    integer err_cnt;
    integer total_checks;

    // -------------------------------------------------------
    // Task: apply reset
    // -------------------------------------------------------
    task apply_reset;
        begin
            rst_n = 0;
            req   = {WIDTH{1'b0}};
            @(posedge clk); #1;
            @(posedge clk); #1;
            rst_n = 1;
            gold_ptr = 0;
        end
    endtask

    // -------------------------------------------------------
    // Task: drive one cycle, check, advance golden model
    // -------------------------------------------------------
    task drive_and_check;
        input [WIDTH-1:0] req_in;
        input [63:0]      cycle_num;
        input [63:0]      case_id;
        reg   [WIDTH-1:0] expected;
        integer gidx, i;
        begin
            req = req_in;
            @(posedge clk); #1;

            compute_golden(req_in, gold_ptr, expected);
            total_checks = total_checks + 1;

            if (grant !== expected) begin
                $display("MISMATCH  case=%0d  cycle=%0d  ptr=%0d  req=%b  got=%b  exp=%b",
                         case_id, cycle_num, gold_ptr, req_in, grant, expected);
                err_cnt = err_cnt + 1;
            end

            // Advance golden pointer
            if (|req_in) begin
                gidx = 0;
                for (i = 0; i < WIDTH; i = i + 1)
                    if (expected[i]) gidx = i;
                gold_ptr = (gidx + 1) % WIDTH;
            end
        end
    endtask

    // -------------------------------------------------------
    // Main stimulus
    // -------------------------------------------------------
    integer i, j, seed;
    reg [WIDTH-1:0] rand_req;

    initial begin
        err_cnt      = 0;
        total_checks = 0;
        seed         = 42;

        // ===================================================
        // RESET
        // ===================================================
        apply_reset;
        $display("=== Reset complete. WIDTH=%0d ===", WIDTH);

        // ===================================================
        // CASE 1: Single requester — req[0] only
        // ===================================================
        $display("--- Case 1: Single requester req[0] ---");
        for (i = 0; i < 8; i = i + 1)
            drive_and_check({{WIDTH-1{1'b0}}, 1'b1}, i, 1);

        apply_reset;

        // ===================================================
        // CASE 2: Two alternating requesters — req[0]/req[1]
        // ===================================================
        $display("--- Case 2: Alternating req[0] / req[1] ---");
        for (i = 0; i < 16; i = i + 1) begin
            if (i[0] == 1'b0)
                drive_and_check({{WIDTH-2{1'b0}}, 2'b01}, i, 2);
            else
                drive_and_check({{WIDTH-2{1'b0}}, 2'b10}, i, 2);
        end

        apply_reset;

        // ===================================================
        // CASE 3: All requesters active — strict round-robin
        // ===================================================
        $display("--- Case 3: All active, strict round-robin ---");
        for (i = 0; i < 3*WIDTH; i = i + 1)
            drive_and_check({WIDTH{1'b1}}, i, 3);

        apply_reset;

        // ===================================================
        // CASE 4: Idle gaps between single requests
        // ===================================================
        $display("--- Case 4: Idle gaps ---");
        for (i = 0; i < WIDTH; i = i + 1) begin
            // assert req[i]
            drive_and_check(1'b1 << i, i*2,   4);
            // idle cycle
            drive_and_check({WIDTH{1'b0}}, i*2+1, 4);
        end

        apply_reset;

        // ===================================================
        // CASE 5: Random patterns (200 cycles)
        // ===================================================
        $display("--- Case 5: Random patterns (200 cycles) ---");
        for (i = 0; i < 200; i = i + 1) begin
            // Simple LFSR-style random using $random
            rand_req = $random(seed) & {WIDTH{1'b1}};
            drive_and_check(rand_req, i, 5);
        end

        // ===================================================
        // RESULTS
        // ===================================================
        $display("==============================================");
        if (err_cnt == 0)
            $display("PASS  — %0d checks, 0 mismatches (WIDTH=%0d)",
                     total_checks, WIDTH);
        else
            $display("FAIL  — %0d mismatches in %0d checks (WIDTH=%0d)",
                     err_cnt, total_checks, WIDTH);
        $display("==============================================");

        #20;
        $finish;
    end

    // -------------------------------------------------------
    // Optional: timeout watchdog
    // -------------------------------------------------------
    initial begin
        #1_000_000;
        $display("TIMEOUT — simulation exceeded limit");
        $finish;
    end

endmodule
